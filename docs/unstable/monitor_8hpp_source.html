34 /**
35 * @ingroup event_handling
36 * @brief Constructed on top of an MPI communicator, this class helps to monitor and communicate events across
37 * multiple processes.
38 *
39 * @details The root process (rank == 0) monitors all other processes. If a process encounters an event, it sends a
40 * message to the root process by calling monitor::report_local_event. The root process then broadcasts this
41 * information to all other processes.
42 *
43 * It can be used to check
44 * - if an event has occurred on any process (monitor::event_on_any_rank) or
45 * - if an event has occurred on all processes (monitor::event_on_all_ranks).
46 *
47 * It uses a duplicate communicator to not interfere with other MPI communications. The communicator is freed in the
48 * `finalize_communications` function (which is called in the destructor if not called before).
49 */
50 class monitor {
51 // Future struct for non-blocking MPI communication.
52 struct future {
53 // MPI request of the non-blocking MPI call.
54 MPI_Request request{};
55
56 // 0 means that no event has occurred, 1 means that an event has occurred.
57 int event = 0;
58 };
59
60 // MPI communicator.
61 mpi::communicator comm;
62
63 // Future objects stored on the root process for local events on non-root processes.
64 std::vector<future> root_futures;
65
66 // MPI request for the broadcasting done on the root process in case an event has occurred on any rank.
67 MPI_Request req_ibcast_any{};
68
69 // MPI request for the broadcasting done on the root process in case an event has occurred on all ranks.
70 MPI_Request req_ibcast_all{};
71
72 // MPI request for the sending done on non-root processes.
73 MPI_Request req_isent{};
74
75 // Set to 1, if a local event has occurred on this process.
76 int local_event = 0;
77
78 // Set to 1, if an event has occurred on any process.
79 int any_event = 0;
80
81 // Set to 1, if an event has occurred on all processes.
82 int all_events = 0;
83
84 // Set to true, if finialize_communications() has been called.
85 bool finalized = false;
86
87 public:
88 /**
89 * @brief Construct a monitor on top of a given mpi::communicator.
90 *
91 * @details The communicator is duplicated to not interfere with other MPI communications.
92 *
93 * The root process (rank == 0) performs a non-blocking receive for every non-root process and waits for a
94 * non-root process to send a message that an event has occurred.
95 *
96 * Non-root processes make two non-blocking broadcast calls and wait for the root process to broadcast a message in
97 * case an event has occurred on any or on all processes.
98 *
99 * @param c mpi::communicator.
100 */
101 monitor(mpi::communicator c) : comm(c.duplicate()) {
102 if (comm.rank() == 0) {
103 root_futures.resize(c.size() - 1);
104 for (int rank = 1; rank < c.size(); ++rank) {
105 MPI_Irecv(&(root_futures[rank - 1].event), 1, MPI_INT, rank, rank, comm.get(), &(root_futures[rank - 1].request));
106 }
107 } else {
108 MPI_Ibcast(&any_event, 1, MPI_INT, 0, comm.get(), &req_ibcast_any);
109 MPI_Ibcast(&all_events, 1, MPI_INT, 0, comm.get(), &req_ibcast_all);
110 }
111 }
112
113 /// Deleted copy constructor.
114 monitor(monitor const &) = delete;
115
116 /// Deleted copy assignment operator.
117 monitor &operator=(monitor const &) = delete;
118
119 /// Destructor calls finalize_communications().
120 ~monitor() { finalize_communications(); }
121
122 /**
123 * @brief Report a local event to the root process (rank == 0).
124 *
125 * @details This function can be called on any process in case a local event has occurred.
126 *
127 * On the root process, it immediately broadcasts to all other processes that an event has occurred and further
128 * checks if all other processes have reported an event as well. If so, it additionally broadcasts to all processes
129 * that an event has occurred on all processes.
130 *
131 * On non-root processes, it sends a message to the root process that a local event has occurred.
132 */
133 void report_local_event() {
134 // prevent sending multiple signals
135 if (local_event or finalized) { return; }
136
137 // a local event has occurred
138 local_event = 1;
139 if (comm.rank() == 0) {
140 // on root process, check all other nodes and perform necessary broadcasts
141 root_check_nodes_and_bcast();
142 } else {
143 // on non-root processes, let the root process know about the local event
144 MPI_Isend(&local_event, 1, MPI_INT, 0, comm.rank(), comm.get(), &req_isent);
145 }
146 }
147
148 /**
149 * @brief Check if an event has occurred on any process.
150 *
151 * @details This function can be called on any process to check if an event has occurred somewhere.
152 *
153 * It returns true, if
154 * - a local event has occurred or
155 * - if an event has occurred on some other process which has already been reported to the root process and
156 * broadcasted to all other processes.
157 *
158 * On the root process (rank == 0), it checks the status of all non-root processes and performs the necessary
159 * broadcasts in case they have not been done yet.
160 *
161 * @return True, if an event has occurred on any process.
162 */
163 [[nodiscard]] bool event_on_any_rank() {
164 // if final_communications() has already been called, any_event == 0 if no event has occurred, otherwise it is 1
165 if (finalized) return any_event;
166
167 // if a local event has occurred, we return true
168 if (local_event) return true;
169
170 // on the root process, we first check the status of all non-root processes, perform the necessary broadcasts and
171 // return true if an event has occurred
172 if (comm.rank() == 0) {
173 root_check_nodes_and_bcast();
174 return any_event;
175 }
176
177 // on non-root processes, we check the status of the corresponding broadcast and return true if an event has
178 // occurred
179 MPI_Status status;
180 int has_received = 0;
181 MPI_Test(&req_ibcast_any, &has_received, &status);
182 return has_received and any_event;
183 }
184
185 /**
186 * @brief Check if an event has occurred on all processes.
187 *
188 * @details This function can be called on any process to check if an event has occurred on all processes.
189 *
190 * It returns true, if an event has occurred on all processes which has already been reported to the root process
191 * and broadcasted to all other processes.
192 *
193 * On the root process (rank == 0), it checks the status of all non-root processes and performs the necessary
194 * broadcasts in case it has not been done yet.
195 *
196 * @return True, if an event has occurred on all processes.
197 */
198 [[nodiscard]] bool event_on_all_ranks() {
199 // if final_communications() has already been called, all_events == 0 if an event has not occurred on every
200 // process, otherwise it is 1
201 if (finalized) return all_events;
202
203 // on the root process, we first check the status of all non-root processes, perform the necessary broadcasts and
204 // return true if an event has occurred on all of them
205 if (comm.rank() == 0) {
206 root_check_nodes_and_bcast();
207 return all_events;
208 }
209
210 // on non-root processes, we check the status of the broadcast and return true if an event has occurred on all
211 // processes
212 MPI_Status status;
213 int has_received = 0;
214 MPI_Test(&req_ibcast_all, &has_received, &status);
215 return has_received and all_events;
216 }
217
218 /**
219 * @brief Finalize all pending communications.
220 *
221 * @details At the end of this function, all MPI communications have been completed and the values of the member
222 * variables will not change anymore due to some member function calls.
223 *
224 * Furthermore, it frees the used communicator.
225 */
226 void finalize_communications() {
227 // prevent multiple calls
228 if (finalized) return;
229
230 if (comm.rank() == 0) {
231 // on root process, wait for all non-root processes to finish their MPI_Isend calls
232 while (root_check_nodes_and_bcast()) {
233 usleep(100); // 100 us (micro seconds)
234 }
235 // and perform broadcasts in case they have not been done yet
236 if (not any_event) { MPI_Ibcast(&any_event, 1, MPI_INT, 0, comm.get(), &req_ibcast_any); }
237 if (not all_events) { MPI_Ibcast(&all_events, 1, MPI_INT, 0, comm.get(), &req_ibcast_all); }
238 } else {
239 // on non-root processes, perform MPI_Isend call in case it has not been done yet
240 if (not local_event) { MPI_Isend(&local_event, 1, MPI_INT, 0, comm.rank(), comm.get(), &req_isent); }
241 }
242
243 // all nodes wait for the broadcasts to be completed
244 MPI_Status status_any, status_all;
245 MPI_Wait(&req_ibcast_any, &status_any);
246 MPI_Wait(&req_ibcast_all, &status_all);
247
248 // free the communicator
249 comm.free();
250 finalized = true;
251 }
252
253 private:
254 // Root process checks the status of all non-root processes, performs necessary broadcasts and returns a boolean
255 // that is true if at least one non-root process has not performed its MPI_Isend call yet.
256 bool root_check_nodes_and_bcast() {
257 EXPECTS(!finalized);
258 EXPECTS(comm.rank() == root);
259 bool any = false;
260 bool all = true;
261 bool finished = true;
262 for (auto &[request, rank_event] : root_futures) {
263 MPI_Status status;
264 int rank_received = 0;
265 MPI_Test(&request, &rank_received, &status);
266 any |= (rank_received and rank_event);
267 all &= (rank_received and rank_event);
268 finished &= rank_received;
269 }
270 if (not any_event and (any or local_event)) {
271 any_event = 1;
272 MPI_Ibcast(&any_event, 1, MPI_INT, 0, comm.get(), &req_ibcast_any);
273 }
274 if (not all_events and all and local_event) {
275 all_events = 1;
276 MPI_Ibcast(&all_events, 1, MPI_INT, 0, comm.get(), &req_ibcast_all);
277 }
278 return not finished;
279 }
280 };
