35 /**
36 * @ingroup err_handling
37 * @brief Constructed on top of an MPI communicator, this class helps to monitor and communicate
38 * exceptions and other errors of individual processes.
39 *
40 * @details The root process (process with rank 0) monitors all other processes. If a process encounters
41 * an error, it sends an emergeny stop request to the root process which forwards it to all the other
42 * processes.
43 */
44 class monitor {
45 // Future struct for the non-blocking send/receive done on the root process.
46 struct future {
47 // MPI request for the non-blocking receive on the root process.
48 MPI_Request request{};
49
50 // 0 means that no error has occurred, 1 means that an error has occurred.
51 int node_stop = 0;
52 };
53
54 // MPI communicator.
55 mpi::communicator com;
56
57 // Future objects stored on the root process for every non-root process.
58 std::vector<future> root_futures;
59
60 // MPI request for broadcasting the emergency stop to all non-root processes.
61 MPI_Request req_ibcast{};
62
63 // MPI request for sending the emergency stop request to the root process.
64 MPI_Request req_isent{};
65
66 // Set to 1, if the process has encountered a local error and requested an emergency stop.
67 int local_stop = 0;
68
69 // Set to 1, if the process has received an emergency stop broadcasted by the root process.
70 int global_stop = 0;
71
72 // Set to true, if finialize_communications() has been called.
73 bool finalized = false;
74
75 public:
76 /**
77 * @brief Construct a monitor on top of a given mpi::communicator.
78 *
79 * @details The root process performs a non-blocking receive for every non-root process and waits for
80 * a non-root process to send an emergency stop request. Non-root processes make a non-blocking broadcast
81 * call and wait for the root process to broadcast any emergency stop request it has received.
82 *
83 * @param c mpi::communicator.
84 */
85 monitor(mpi::communicator c) : com(c) {
86 if (com.rank() == 0) {
87 root_futures.resize(c.size() - 1);
88 for (int rank = 1; rank < c.size(); ++rank) {
89 MPI_Irecv(&(root_futures[rank - 1].node_stop), 1, MPI_INT, rank, 0, MPI_COMM_WORLD, &(root_futures[rank - 1].request));
90 }
91 } else {
92 MPI_Ibcast(&global_stop, 1, MPI_INT, 0, MPI_COMM_WORLD, &req_ibcast);
93 }
94 }
95
96 /// Deleted copy constructor.
97 monitor(monitor const &) = delete;
98
99 /// Deleted copy assignment operator.
100 monitor &operator=(monitor const &) = delete;
101
102 /// Destructor calls finalize_communications().
103 ~monitor() { finalize_communications(); }
104
105 /**
106 * @brief Request an emergency stop.
107 *
108 * @details This function can be called on any process in case a local error has occurred. On the
109 * root process, it sets its `local_stop` and `global_stop` members to 1 and broadcasts `global_stop`
110 * to all non-root processes. On non-root processes, it sets `local_stop` to 1 and sends it to the
111 * root process.
112 */
113 void request_emergency_stop() {
114 EXPECTS(!finalized);
115 // prevent sending multiple signals
116 if (local_stop) { return; }
117
118 // an error has occurred
119 local_stop = 1;
120 if (com.rank() == 0) {
121 // root broadcasts the global_stop variable
122 global_stop = 1;
123 MPI_Ibcast(&global_stop, 1, MPI_INT, 0, MPI_COMM_WORLD, &req_ibcast);
124 } else {
125 // non-root sends the local_stop variable to root
126 MPI_Isend(&local_stop, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &req_isent);
127 }
128 }
129
130 /**
131 * @brief Check if an emergency stop has been requested.
132 *
133 * @details This function can be called on any process to check if an emergency has occurred somewhere.
134 * It first checks, if its `local_stop` or `global_stop` members are set to 1 and returns `true` in case
135 * one of them is. Otherwise, on the root process, it calls `root_check_nodes_and_bcast()` to check if
136 * some other process has sent an emergency message and to possibly forward the received signal.
137 * On non-root processes, it checks if the root process has broadcasted an emergency stop, which it has
138 * received from some other process.
139 *
140 * @return True, if an emergency stop has been requested. Otherwise, it returns false.
141 */
142 [[nodiscard]] bool emergency_occured() {
143 // if final_communications() has already been called, global_stop == 0 if no error has occurred, otherwise it is 1
144 if (finalized) return global_stop;
145
146 // either a local error has occurred or some other process has requested an emergency stop
147 if (global_stop or local_stop) return true;
148
149 if (com.rank() == 0) {
150 // root checks if some other process has requested an emergency stop
151 root_check_nodes_and_bcast();
152 }
153 return global_stop;
154 }
155
156 /**
157 * @brief Finalize all pending communications.
158 *
159 * @details At the end of this function, all processes have completed their work or have had a local
160 * emergency stop. The member `global_stop` is guaranteed to be the same on all processes when this
161 * function returns.
162 */
163 void finalize_communications() {
164 if (finalized) return;
165 if (com.rank() == 0) {
166 // root just listens to the other processes and bcasts the global_stop until everyone is done
167 while (root_check_nodes_and_bcast()) { usleep(100); } // 100 us (micro seconds)
168 // all others node have finished
169 // if the root has never emitted the ibcast, we do it now
170 if (not global_stop) { MPI_Ibcast(&global_stop, 1, MPI_INT, 0, MPI_COMM_WORLD, &req_ibcast); }
171 } else {
172 // on non-root node: either Isend was done when local_stop was set to 1 during request_emergency_stop,
173 // or it has to happen now, i.e, work is done, and fine.
174 if (not local_stop) { MPI_Isend(&local_stop, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &req_isent); }
175 }
176 // all nodes wait for the ibcast of the global_stop to be complete
177 MPI_Status status;
178 MPI_Wait(&req_ibcast, &status);
179 finalized = true;
180 }
181
182 private:
183 /**
184 * @brief Check if any non-root process has sent a stop request. If so, broadcast to all other processes
185 * in case it has not been done yet.
186 *
187 * @return True, if at least one process has not finished the `MPI_Isend` of the `local_stop` variable to
188 * the root process. Otherwise, it returns false.
189 */
190 bool root_check_nodes_and_bcast() {
191 EXPECTS(!finalized);
192 EXPECTS(com.rank() == 0);
193 // loop over all non-root processes
194 bool some_nodes_are_still_running = false;
195 for (auto &[request, node_stop] : root_futures) {
196 // check for an emergency stop request
197 MPI_Status status;
198 int comm_received = 0;
199 MPI_Test(&request, &comm_received, &status);
200 // for the first time an emergency stop has been requested -> root calls request_emergency_stop()
201 // to broadcast to all other processes
202 if (comm_received and (not global_stop) and node_stop) request_emergency_stop(); // the root requires the stop now. It also stops itself...
203 some_nodes_are_still_running |= (not comm_received);
204 }
205 return some_nodes_are_still_running;
206 }
207 };
