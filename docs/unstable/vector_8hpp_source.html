54 // Check if a vector is of the same size on all processes.
55 template <typename T> bool same_size(std::vector<T> const &v, communicator c) {
56 auto size = v.size();
57 auto max_size = mpi::all_reduce(size, c, MPI_MAX);
58 auto min_size = mpi::all_reduce(size, c, MPI_MIN);
59 return min_size == max_size;
60 }

71 /**
72 * @brief Implementation of an MPI broadcast for a std::vector.
73 *
74 * @details If mpi::has_mpi_type<T> is true then the vector is broadcasted using a simple `MPI_Bcast`. Otherwise, the
75 * generic mpi::broadcast is called for each element of the vector.
76 *
77 * On non-root processes, the vector is always resized to match the size of the vector on the root process. If the
78 * root process is empty, no broadcast is performed.
79 *
80 * @tparam T Value type of the vector.
81 * @param v std::vector to broadcast.
82 * @param c mpi::communicator.
83 * @param root Rank of the root process.
84 */
85 template <typename T> void mpi_broadcast(std::vector<T> &v, communicator c = {}, int root = 0) {
86 auto s = v.size();
87 broadcast(s, c, root);
88 if (c.rank() != root) v.resize(s);
89 if constexpr (has_mpi_type<T>) {
90 if (s != 0) MPI_Bcast(v.data(), v.size(), mpi_type<T>::get(), root, c.get());
91 } else {
92 for (auto &x : v) broadcast(x, c, root);
93 }
94 }

96 /**
97 * @brief Implementation of an in-place MPI reduce for a std::vector.
98 *
99 * @details If mpi::has_mpi_type<T> is true then the vector is reduced using a simple `MPI_Reduce` or `MPI_Allreduce`.
100 * Otherwise, the specialized `mpi_reduce_in_place` is called for each element of the vector.
101 *
102 * It throws an exception if the vectors are not of the same size. If the vectors are empty, it does nothing.
103 *
104 * @tparam T Value type of the vector.
105 * @param v std::vector to reduce.
106 * @param c mpi::communicator.
107 * @param root Rank of the root process.
108 * @param all Should all processes receive the result of the reduction.
109 * @param op `MPI_Op` used in the reduction.
110 */
111 template <typename T> void mpi_reduce_in_place(std::vector<T> &v, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
112 // check the sizes of all vectors
113 if (!detail::same_size(v, c)) throw std::runtime_error{"mpi_reduce_in_place for vectors only works with vectors of the same size\n"};
114 if (v.size() == 0) return;
115
116 if constexpr (has_mpi_type<T>) {
117 if (!all)
118 MPI_Reduce((c.rank() == root ? MPI_IN_PLACE : v.data()), v.data(), v.size(), mpi_type<T>::get(), op, root, c.get());
119 else
120 MPI_Allreduce(MPI_IN_PLACE, v.data(), v.size(), mpi_type<T>::get(), op, c.get());
121 } else {
122 for (auto &x : v) mpi_reduce_in_place(v, c, root, all);
123 }
124 }

126 /**
127 * @brief Implementation of an MPI reduce for a std::vector.
128 *
129 * @details If mpi::has_mpi_type<T> is true then the vector is reduced using a simple `MPI_Reduce` or `MPI_Allreduce`
130 * (in this case, mpi::regular_t<T> has to be the same as T). Otherwise, the generic mpi::reduce is called for each
131 * element of the vector.
132 *
133 * It throws an exception if the vectors are not of the same size. If the vectors are empty, it does nothing.
134 *
135 * @tparam T Value type of the vector.
136 * @param v std::vector to reduce.
137 * @param c mpi::communicator.
138 * @param root Rank of the root process.
139 * @param all Should all processes receive the result of the reduction.
140 * @param op `MPI_Op` used in the reduction.
141 * @return std::vector containing the result of each individual reduction.
142 */
143 template <typename T>
144 std::vector<regular_t<T>> mpi_reduce(std::vector<T> const &v, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
145 // check the size of all vectors
146 if (!detail::same_size(v, c)) throw std::runtime_error{"mpi_reduce for vectors only works with vectors of the same size\n"};
147 if (v.size() == 0) return {};
148
149 // perform the reduction for every element of the vector
150 auto s = v.size();
151 if constexpr (has_mpi_type<T>) {
152 static_assert(std::is_same_v<regular_t<T>, T>, "Internal error");
153 std::vector<T> res(s);
154 if (!all)
155 MPI_Reduce((void *)v.data(), res.data(), s, mpi_type<T>::get(), op, root, c.get());
156 else
157 MPI_Allreduce((void *)v.data(), res.data(), s, mpi_type<T>::get(), op, c.get());
158 return res;
159 } else {
160 std::vector<regular_t<T>> r;
161 r.reserve(s);
162 for (size_t i = 0; i < s; ++i) r.push_back(reduce(v[i], c, root, all, op));
163 return r;
164 }
165 }

167 /**
168 * @brief Implementation of an MPI scatter for a std::vector.
169 *
170 * @details If mpi::has_mpi_type<T> is true then the vector is scattered as evenly as possible across the processes in
171 * the communicator using a simple `MPI_Scatterv`.
172 *
173 * @tparam T Value type of the vector.
174 * @param v std::vector to scatter.
175 * @param c mpi::communicator.
176 * @param root Rank of the root process.
177 * @return std::vector containing the result of the scatter operation.
178 */
179 template <typename T> std::vector<T> mpi_scatter(std::vector<T> const &v, communicator c = {}, int root = 0) {
180 auto s = v.size();
181
182 // return an empty vector if size is 0
183 mpi::broadcast(s, c, root);
184 if (s == 0) return {};
185
186 // arguments for the MPI call
187 auto sendcounts = std::vector<int>(c.size()); // number of elements sent to each process
188 auto displs = std::vector<int>(c.size() + 1, 0); // displacements given in number of elements not in bytes
189 int recvcount = chunk_length(s, c.size(), c.rank()); // number of elements received by the calling process
190 for (int r = 0; r < c.size(); ++r) {
191 sendcounts[r] = chunk_length(s, c.size(), r);
192 displs[r + 1] = sendcounts[r] + displs[r];
193 }
194
195 // do the scattering
196 std::vector<T> res(recvcount);
197 if constexpr (has_mpi_type<T>) {
198 MPI_Scatterv((void *)v.data(), &sendcounts[0], &displs[0], mpi_type<T>::get(), (void *)res.data(), recvcount, mpi_type<T>::get(), root,
199 c.get());
200 } else {
201 std::copy(cbegin(v) + displs[c.rank()], cbegin(v) + displs[c.rank() + 1], begin(res));
202 }
203
204 return res;
205 }

207 /**
208 * @brief Implementation of an MPI gather for a std::vector.
209 *
210 * @details If mpi::has_mpi_type<T> is true then the vector is gathered using a simple `MPI_Gatherv` or
211 * `MPI_Allgatherv`. Otherwise, each process broadcasts its elements to all other processes which implies that
212 * `all == true` is required in this case.
213 *
214 * @tparam T Value type of the vector.
215 * @param v std::vector to gather.
216 * @param c mpi::communicator.
217 * @param root Rank of the root process.
218 * @param all Should all processes receive the result of the reduction.
219 * @return std::vector containing the result of the gather operation.
220 */
221 template <typename T> std::vector<T> mpi_gather(std::vector<T> const &v, communicator c = {}, int root = 0, bool all = false) {
222 long s = mpi::all_reduce(v.size(), c);
223
224 // return an empty vector if size is 0
225 if (s == 0) return {};
226
227 // arguments for the MPI call
228 auto mpi_ty = mpi_type<int>::get();
229 auto recvcounts = std::vector<int>(c.size()); // number of elements received from each process
230 auto displs = std::vector<int>(c.size() + 1, 0); // displacements given in number of elements not in bytes
231 int sendcount = v.size(); // number of elements sent by the calling process
232 if (!all)
233 MPI_Gather(&sendcount, 1, mpi_ty, &recvcounts[0], 1, mpi_ty, root, c.get());
234 else
235 MPI_Allgather(&sendcount, 1, mpi_ty, &recvcounts[0], 1, mpi_ty, c.get());
236
237 for (int r = 0; r < c.size(); ++r) displs[r + 1] = recvcounts[r] + displs[r];
238
239 // do the gathering
240 std::vector<T> res((all || (c.rank() == root) ? s : 0));
241 if constexpr (has_mpi_type<T>) {
242 if (!all)
243 MPI_Gatherv((void *)v.data(), sendcount, mpi_type<T>::get(), (void *)res.data(), &recvcounts[0], &displs[0], mpi_type<T>::get(), root,
244 c.get());
245 else
246 MPI_Allgatherv((void *)v.data(), sendcount, mpi_type<T>::get(), (void *)res.data(), &recvcounts[0], &displs[0], mpi_type<T>::get(), c.get());
247 } else {
248 if (!all)
249 throw std::runtime_error{"mpi_gather for custom types only implemented with 'all = true'\n"};
250 else {
251 for (int r = 0; r < c.size(); ++r) {
252 for (auto i = displs[r]; i < displs[r + 1]; ++i) {
253 if (c.rank() == r) res[i] = v[i - displs[r]];
254 mpi::broadcast(res[i], c, r);
255 }
256 }
257 }
258 }
259 return res;
260 }
