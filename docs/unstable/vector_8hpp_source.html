43 /**
44 * @brief Implementation of an MPI broadcast for a std::vector.
45 *
46 * @details If mpi::has_mpi_type<T> is true then the vector is broadcasted using a simple `MPI_Bcast`. Otherwise,
47 * the generic mpi::broadcast is called for each element of the vector.
48 *
49 * @tparam T Value type of the vector.
50 * @param v std::vector to broadcast.
51 * @param c mpi::communicator.
52 * @param root Rank of the root process.
53 */
54 template <typename T> void mpi_broadcast(std::vector<T> &v, communicator c = {}, int root = 0) {
55 auto s = v.size();
56 broadcast(s, c, root);
57 if (c.rank() != root) v.resize(s);
58 if constexpr (has_mpi_type<T>) {
59 if (s != 0) MPI_Bcast(v.data(), v.size(), mpi_type<T>::get(), root, c.get());
60 } else {
61 for (auto &x : v) broadcast(x, c, root);
62 }
63 }

65 /**
66 * @brief Implementation of an in-place MPI reduce for a std::vector.
67 *
68 * @details If mpi::has_mpi_type<T> is true then the vector is reduced using a simple `MPI_Reduce` or `MPI_Allreduce`.
69 * Otherwise, the specialized `mpi_reduce_in_place` is called for each element of the vector.
70 *
71 * @tparam T Value type of the vector.
72 * @param v std::vector to reduce.
73 * @param c mpi::communicator.
74 * @param root Rank of the root process.
75 * @param all Should all processes receive the result of the reduction.
76 * @param op `MPI_Op` used in the reduction.
77 */
78 template <typename T> void mpi_reduce_in_place(std::vector<T> &v, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
79 if (v.size() == 0) return;
80 if constexpr (has_mpi_type<T>) {
81 if (!all)
82 MPI_Reduce((c.rank() == root ? MPI_IN_PLACE : v.data()), v.data(), v.size(), mpi_type<T>::get(), op, root, c.get());
83 else
84 MPI_Allreduce(MPI_IN_PLACE, v.data(), v.size(), mpi_type<T>::get(), op, c.get());
85 } else {
86 for (auto &x : v) mpi_reduce_in_place(v, c, root, all);
87 }
88 }

111 /**
112 * @brief Implementation of an MPI reduce for a std::vector.
113 *
114 * @details If mpi::has_mpi_type<T> is true then the vector is reduced using a simple `MPI_Reduce` or `MPI_Allreduce`
115 * (in this case, mpi::regular_t<T> has to be the same as T). Otherwise, the generic mpi::reduce is called for each
116 * element of the vector.
117 *
118 * @tparam T Value type of the vector.
119 * @param v std::vector to reduce.
120 * @param c mpi::communicator.
121 * @param root Rank of the root process.
122 * @param all Should all processes receive the result of the reduction.
123 * @param op `MPI_Op` used in the reduction.
124 * @return std::vector containing the result of each individual reduction.
125 */
126 template <typename T>
127 std::vector<regular_t<T>> mpi_reduce(std::vector<T> const &v, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
128 auto s = v.size();
129
130 // check if all vectors are of the same size, otherwise abort
131 if (all) {
132 auto max_size = mpi_reduce(s, c, root, all, MPI_MAX);
133 if (s != max_size) {
134 std::cerr << "Cannot all_reduce vectors of different sizes\n";
135 std::abort();
136 }
137 }
138
139 // return an empty vector if size is 0
140 if (s == 0) return {};
141
142 // perform the reduction for every element of the vector
143 if constexpr (has_mpi_type<T>) {
144 static_assert(std::is_same_v<regular_t<T>, T>, "Internal error");
145 std::vector<T> res(s);
146 if (!all)
147 MPI_Reduce((void *)v.data(), res.data(), s, mpi_type<T>::get(), op, root, c.get());
148 else
149 MPI_Allreduce((void *)v.data(), res.data(), s, mpi_type<T>::get(), op, c.get());
150 return res;
151 } else {
152 std::vector<regular_t<T>> r;
153 r.reserve(s);
154 for (size_t i = 0; i < s; ++i) r.push_back(reduce(v[i], c, root, all, op));
155 return r;
156 }
157 }

159 /**
160 * @brief Implementation of an MPI scatter for a std::vector.
161 *
162 * @details If mpi::has_mpi_type<T> is true then the vector is scattered as evenly as possible across the processes
163 * in the communicator using a simple `MPI_Scatterv`.
164 *
165 * @tparam T Value type of the vector.
166 * @param v std::vector to scatter.
167 * @param c mpi::communicator.
168 * @param root Rank of the root process.
169 * @return std::vector containing the result of the scatter operation.
170 */
171 template <typename T> std::vector<T> mpi_scatter(std::vector<T> const &v, communicator c = {}, int root = 0) {
172 auto s = v.size();
173
174 // return an empty vector if size is 0
175 if (s == 0) return {};
176
177 // arguments for the MPI call
178 auto sendcounts = std::vector<int>(c.size()); // number of elements sent to each process
179 auto displs = std::vector<int>(c.size() + 1, 0); // displacements given in number of elements not in bytes
180 int recvcount = chunk_length(s, c.size(), c.rank()); // number of elements received by the calling process
181 for (int r = 0; r < c.size(); ++r) {
182 sendcounts[r] = chunk_length(s, c.size(), r);
183 displs[r + 1] = sendcounts[r] + displs[r];
184 }
185
186 // do the scattering
187 std::vector<T> res(recvcount);
188 if constexpr (has_mpi_type<T>) {
189 MPI_Scatterv((void *)v.data(), &sendcounts[0], &displs[0], mpi_type<T>::get(), (void *)res.data(), recvcount, mpi_type<T>::get(), root,
190 c.get());
191 } else {
192 std::copy(cbegin(v) + displs[c.rank()], cbegin(v) + displs[c.rank() + 1], begin(res));
193 }
194
195 return res;
196 }

198 /**
199 * @brief Implementation of an MPI gather for a std::vector.
200 *
201 * @details If mpi::has_mpi_type<T> is true then the vector is gathered using a simple `MPI_Gatherv` or `MPI_Allgatherv`.
202 * Otherwise, each process broadcasts its elements to all other processes which implies that `all == true` is required
203 * in this case.
204 *
205 * @tparam T Value type of the vector.
206 * @param v std::vector to gather.
207 * @param c mpi::communicator.
208 * @param root Rank of the root process.
209 * @param all Should all processes receive the result of the reduction.
210 * @return std::vector containing the result of the gather operation.
211 */
212 template <typename T> std::vector<T> mpi_gather(std::vector<T> const &v, communicator c = {}, int root = 0, bool all = false) {
213 long s = mpi_reduce(v.size(), c, root, all);
214
215 // return an empty vector if size is 0
216 if (s == 0) return {};
217
218 // arguments for the MPI call
219 auto mpi_ty = mpi_type<int>::get();
220 auto recvcounts = std::vector<int>(c.size()); // number of elements received from each process
221 auto displs = std::vector<int>(c.size() + 1, 0); // displacements given in number of elements not in bytes
222 int sendcount = v.size(); // number of elements sent by the calling process
223 if (!all)
224 MPI_Gather(&sendcount, 1, mpi_ty, &recvcounts[0], 1, mpi_ty, root, c.get());
225 else
226 MPI_Allgather(&sendcount, 1, mpi_ty, &recvcounts[0], 1, mpi_ty, c.get());
227
228 for (int r = 0; r < c.size(); ++r) displs[r + 1] = recvcounts[r] + displs[r];
229
230 // do the gathering
231 std::vector<T> res((all || (c.rank() == root) ? s : 0));
232 if constexpr (has_mpi_type<T>) {
233 if (!all)
234 MPI_Gatherv((void *)v.data(), sendcount, mpi_type<T>::get(), (void *)res.data(), &recvcounts[0], &displs[0], mpi_type<T>::get(), root,
235 c.get());
236 else
237 MPI_Allgatherv((void *)v.data(), sendcount, mpi_type<T>::get(), (void *)res.data(), &recvcounts[0], &displs[0], mpi_type<T>::get(), c.get());
238 } else {
239 if (!all)
240 throw std::runtime_error{"mpi_gather for custom types only implemented with 'all = true'\n"};
241 else {
242 for (int r = 0; r < c.size(); ++r) {
243 for (auto i = displs[r]; i < displs[r + 1]; ++i) {
244 if (c.rank() == r) res[i] = v[i - displs[r]];
245 mpi::broadcast(res[i], c, r);
246 }
247 }
248 }
249 }
250 return res;
251 }
