117 template <typename... Ts> [[nodiscard]] MPI_Datatype get_mpi_type(std::tuple<Ts...> tup) {
118 static constexpr int N = sizeof...(Ts);
119 std::array<MPI_Datatype, N> types = {mpi_type<std::remove_reference_t<Ts>>::get()...};
120
121 // the number of elements per type (we want 1 per type)
122 std::array<int, N> blocklen;
123 for (int i = 0; i < N; ++i) { blocklen[i] = 1; }
124
125 // displacements of the blocks in bytes w.r.t. to the memory address of the first block
126 std::array<MPI_Aint, N> disp;
127 // initialize displacement array from the tuple element addresses
128 []<size_t... Is>(std::index_sequence<Is...>, auto &t, MPI_Aint *d) {
129 ((d[Is] = (char *)&std::get<Is>(t) - (char *)&std::get<0>(t)), ...);
130 // account for non-trivial memory layouts of the tuple elements
131 auto min_el = *std::min_element(d, d + sizeof...(Ts));
132 ((d[Is] -= min_el), ...);
133 }(std::index_sequence_for<Ts...>{}, tup, disp.data());
134
135 // create and return MPI datatype
136 MPI_Datatype cty{};
137 check_mpi_call(MPI_Type_create_struct(N, blocklen.data(), disp.data(), types.data(), &cty), "MPI_Type_create_struct");
138 check_mpi_call(MPI_Type_commit(&cty), "MPI_Type_commit");
139 return cty;
140 }

185 // Archive helper class to obtain MPI custom type info using references to class members.
186 struct mpi_archive {
187 std::vector<int> block_lengths{};
188 std::vector<MPI_Aint> displacements{};
189 std::vector<MPI_Datatype> types{};
190 MPI_Aint base_address{};
191
192 // Constructor sets the base address of the object.
193 explicit mpi_archive(const void *base) { MPI_Get_address(base, &base_address); }
194
195 // Overloaded operator& to process members to set the block lengths, displacements and MPI types.
196 template <typename T>
197 requires(has_mpi_type<T>)
198 mpi_archive &operator&(const T &member) {
199 types.push_back(mpi_type<T>::get());
200 MPI_Aint address{};
201 MPI_Get_address(&member, &address);
202 displacements.push_back(MPI_Aint_diff(address, base_address));
203 block_lengths.push_back(1);
204 return *this;
205 }
206 };

227 template <Serializable T> [[nodiscard]] MPI_Datatype get_mpi_type(const T &obj) {
228 detail::mpi_archive ar(&obj);
229 obj.serialize(ar);
230 MPI_Datatype mpi_type{};
231 MPI_Type_create_struct(static_cast<int>(ar.block_lengths.size()), ar.block_lengths.data(), ar.displacements.data(), ar.types.data(), &mpi_type);
232 MPI_Type_commit(&mpi_type);
233 return mpi_type;
