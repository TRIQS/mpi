34 /**
35 * @ingroup mpi_essentials
36 * @brief C++ wrapper around `MPI_Comm` providing various convenience functions.
37 *
38 * @details It stores an `MPI_Comm` object as its only member which by default is set to `MPI_COMM_WORLD`.
39 * Note that copying the communicator simply copies the `MPI_Comm` object, without calling `MPI_Comm_dup`.
40 *
41 * All functions that make direct calls to the MPI C library throw an exception in case the call fails.
42 */
43 class communicator {
44 // Wrapped `MPI_Comm` object.
45 MPI_Comm _com = MPI_COMM_WORLD;
46
47 public:
48 /// Construct a communicator with `MPI_COMM_WORLD`.
49 communicator() = default;
50
51 /**
52 * @brief Construct a communicator with a given `MPI_Comm` object.
53 * @details The `MPI_Comm` object is copied without calling `MPI_Comm_dup`.
54 */
55 communicator(MPI_Comm c) : _com(c) {}
56
57 /// Get the wrapped `MPI_Comm` object.
58 [[nodiscard]] MPI_Comm get() const noexcept { return _com; }
59
60 /**
61 * @brief Get the rank of the calling process in the communicator.
62 * @return The result of `MPI_Comm_rank` if mpi::has_env is true, otherwise 0.
63 */
64 [[nodiscard]] int rank() const {
65 if (has_env) {
66 int num = 0;
67 check_mpi_call(MPI_Comm_rank(_com, &num), "MPI_Comm_rank");
68 return num;
69 } else
70 return 0;
71 }
72
73 /**
74 * @brief Get the size of the communicator.
75 * @return The result of `MPI_Comm_size` if mpi::has_env is true, otherwise 1.
76 */
77 [[nodiscard]] int size() const {
78 if (has_env) {
79 int num = 0;
80 check_mpi_call(MPI_Comm_size(_com, &num), "MPI_Comm_size");
81 return num;
82 } else
83 return 1;
84 }
85
86 /**
87 * @brief Split the communicator into disjoint subgroups.
88 *
89 * @details Calls `MPI_Comm_split` with the given color and key arguments. See the MPI documentation for more details,
90 * e.g. <a href="https://docs.open-mpi.org/en/v5.0.x/man-openmpi/man3/MPI_Comm_split.3.html">open-mpi docs</a>.
91 *
92 * @warning This allocates a new communicator object. Make sure to call `free` on the returned communicator when
93 * it is no longer needed.
94 *
95 * @return If mpi::has_env is true, return the split `MPI_Comm` object wrapped in a new mpi::communicator, otherwise
96 * return a default constructed mpi::communicator.
97 */
98 [[nodiscard]] communicator split(int color, int key = 0) const {
99 if (has_env) {
100 communicator c;
101 check_mpi_call(MPI_Comm_split(_com, color, key, &c._com), "MPI_Comm_split");
102 return c;
103 } else
104 return {};
105 }
106
107 /**
108 * @brief Duplicate the communicator.
109 *
110 * @details Calls `MPI_Comm_dup` to duplicate the communicator. See the MPI documentation for more details, e.g.
111 * <a href="https://docs.open-mpi.org/en/v5.0.x/man-openmpi/man3/MPI_Comm_dup.3.html">open-mpi docs</a>.
112 *
113 * @warning This allocates a new communicator object. Make sure to call `free` on the returned communicator when
114 * it is no longer needed.
115 *
116 * @return If mpi::has_env is true, return the duplicated `MPI_Comm` object wrapped in a new mpi::communicator,
117 * otherwise return a default constructed mpi::communicator.
118 */
119 [[nodiscard]] communicator duplicate() const {
120 if (has_env) {
121 communicator c;
122 check_mpi_call(MPI_Comm_dup(_com, &c._com), "MPI_Comm_dup");
123 return c;
124 } else
125 return {};
126 }
127
128 /**
129 * @brief Free the communicator.
130 *
131 * @details Calls `MPI_Comm_free` to mark the communicator for deallocation. See the MPI documentation for more
132 * details, e.g. <a href="https://docs.open-mpi.org/en/v5.0.x/man-openmpi/man3/MPI_Comm_free.3.html">open-mpi docs
133 * </a>.
134 *
135 * Does nothing, if mpi::has_env is false.
136 */
137 void free() {
138 if (has_env) { check_mpi_call(MPI_Comm_free(&_com), "MPI_Comm_free"); }
139 }
140
141 /**
142 * @brief If mpi::has_env is true, `MPI_Abort` is called with the given error code, otherwise std::abort is called.
143 * @param error_code The error code to pass to `MPI_Abort`.
144 */
145 void abort(int error_code) const {
146 if (has_env)
147 check_mpi_call(MPI_Abort(_com, error_code), "MPI_Abort");
148 else
149 std::abort();
150 }

152#ifdef BOOST_MPI_HPP
153 // Conversion to and from boost communicator, Keep for backward compatibility
154 inline operator boost::mpi::communicator() const { return boost::mpi::communicator(_com, boost::mpi::comm_duplicate); }
155 inline communicator(boost::mpi::communicator c) : _com(c) {}

158 /**
159 * @brief Barrier synchronization.
160 *
161 * @details Does nothing if mpi::has_env is false. Otherwise, it either uses a blocking `MPI_Barrier`
162 * (if the given argument is 0) or a non-blocking `MPI_Ibarrier` call. The given parameter determines
163 * in milliseconds how often each process calls `MPI_Test` to check if all processes have reached the barrier.
164 * This can considerably reduce the CPU load:
165 * - 1 msec ~ 1% cpu load
166 * - 10 msec ~ 0.5% cpu load
167 * - 100 msec ~ 0.01% cpu load
168 *
169 * For a very unbalanced load that takes a long time to finish, 1000 msec is a good choice.
170 *
171 * @param poll_msec The polling interval in milliseconds. If set to 0, a simple `MPI_Barrier` call is used.
172 */
173 void barrier(long poll_msec = 1) const {
174 if (has_env) {
175 if (poll_msec == 0) {
176 check_mpi_call(MPI_Barrier(_com), "MPI_Barrier");
177 } else {
178 MPI_Request req{};
179 int flag = 0;
180 check_mpi_call(MPI_Ibarrier(_com, &req), "MPI_Ibarrier");
181 while (!flag) {
182 check_mpi_call(MPI_Test(&req, &flag, MPI_STATUS_IGNORE), "MPI_Test");
183 usleep(poll_msec * 1000);
184 }
185 }
186 }
187 }
188 };
