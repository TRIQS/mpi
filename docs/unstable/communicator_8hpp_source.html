33 /**
34 * @ingroup mpi_essentials
35 * @brief C++ wrapper around `MPI_Comm` providing various convenience functions.
36 *
37 * @details It stores an `MPI_Comm` object as its only member which by default is set to `MPI_COMM_WORLD`.
38 * Note that copying the communicator simply copies the `MPI_Comm` object, without calling `MPI_Comm_dup`.
39 */
40 class communicator {
41 // Wrapped `MPI_Comm` object.
42 MPI_Comm _com = MPI_COMM_WORLD;
43
44 public:
45 /// Construct a communicator with `MPI_COMM_WORLD`.
46 communicator() = default;
47
48 /**
49 * @brief Construct a communicator with a given `MPI_Comm` object.
50 * @details The `MPI_Comm` object is copied without calling `MPI_Comm_dup`.
51 */
52 communicator(MPI_Comm c) : _com(c) {}
53
54 /// Get the wrapped `MPI_Comm` object.
55 [[nodiscard]] MPI_Comm get() const noexcept { return _com; }
56
57 /**
58 * @brief Get the rank of the calling process in the communicator.
59 * @return The result of `MPI_Comm_rank` if mpi::has_env is true, otherwise 0.
60 */
61 [[nodiscard]] int rank() const {
62 if (has_env) {
63 int num = 0;
64 MPI_Comm_rank(_com, &num);
65 return num;
66 } else
67 return 0;
68 }
69
70 /**
71 * @brief Get the size of the communicator.
72 * @return The result of `MPI_Comm_size` if mpi::has_env is true, otherwise 1.
73 */
74 [[nodiscard]] int size() const {
75 if (has_env) {
76 int num = 0;
77 MPI_Comm_size(_com, &num);
78 return num;
79 } else
80 return 1;
81 }
82
83 /**
84 * @brief Split the communicator into disjoint subgroups.
85 *
86 * @details Calls `MPI_Comm_split` with the given color and key arguments. See the MPI documentation for more details,
87 * e.g. <a href="https://docs.open-mpi.org/en/v5.0.x/man-openmpi/man3/MPI_Comm_split.3.html">open-mpi docs</a>.
88 *
89 * @return If mpi::has_env is true, return the split `MPI_Comm` object wrapped in a new mpi::communicator, otherwise
90 * return a default constructed mpi::communicator.
91 */
92 [[nodiscard]] communicator split(int color, int key = 0) const {
93 if (has_env) {
94 communicator c;
95 MPI_Comm_split(_com, color, key, &c._com);
96 return c;
97 } else
98 return {};
99 }
100
101 /**
102 * @brief If mpi::has_env is true, `MPI_Abort` is called with the given error code, otherwise std::abort is called.
103 * @param error_code The error code to pass to `MPI_Abort`.
104 */
105 void abort(int error_code) {
106 if (has_env)
107 MPI_Abort(_com, error_code);
108 else
109 std::abort();
110 }

118 /**
119 * @brief Barrier synchronization.
120 *
121 * @details Does nothing if mpi::has_env is false. Otherwise, it either uses a blocking `MPI_Barrier`
122 * (if the given argument is 0) or a non-blocking `MPI_Ibarrier` call. The given parameter determines
123 * in milliseconds how often each process calls `MPI_Test` to check if all processes have reached the barrier.
124 * This can considerably reduce the CPU load:
125 * - 1 msec ~ 1% cpu load
126 * - 10 msec ~ 0.5% cpu load
127 * - 100 msec ~ 0.01% cpu load
128 *
129 * For a very unbalanced load that takes a long time to finish, 1000 msec is a good choice.
130 *
131 * @param poll_msec The polling interval in milliseconds. If set to 0, a simple `MPI_Barrier` call is used.
132 */
133 void barrier(long poll_msec = 1) {
134 if (has_env) {
135 if (poll_msec == 0) {
136 MPI_Barrier(_com);
137 } else {
138 MPI_Request req{};
139 int flag = 0;
140 MPI_Ibarrier(_com, &req);
141 while (!flag) {
142 MPI_Test(&req, &flag, MPI_STATUS_IGNORE);
143 usleep(poll_msec * 1000);
144 }
145 }
146 }
147 }
148 };
