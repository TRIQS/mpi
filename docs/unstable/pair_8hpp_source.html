37 /**
38 * @brief Implementation of an MPI broadcast for a std::pair.
39 *
40 * @details Simply calls the generic mpi::broadcast for the first and second element of the pair.
41 *
42 * @tparam T1 Type of the first element of the pair.
43 * @tparam T2 Type of the second element of the pair.
44 * @param p std::pair to broadcast.
45 * @param c mpi::communicator.
46 * @param root Rank of the root process.
47 */
48 template <typename T1, typename T2> void mpi_broadcast(std::pair<T1, T2> &p, communicator c = {}, int root = 0) {
49 broadcast(p.first, c, root);
50 broadcast(p.second, c, root);
51 }
52
53 /**
54 * @brief Implementation of an MPI reduce for a std::pair.
55 *
56 * @details Simply calls the generic mpi::reduce for the first and second element of the pair.
57 *
58 * @tparam T1 Type of the first element of the pair.
59 * @tparam T2 Type of the second element of the pair.
60 * @param p std::pair to be reduced.
61 * @param c mpi::communicator.
62 * @param root Rank of the root process.
63 * @param all Should all processes receive the result of the reduction.
64 * @param op `MPI_Op` used in the reduction.
65 * @return std::pair<T1, T2> containing the result of each individual reduction.
66 */
67 template <typename T1, typename T2>
68 auto mpi_reduce(std::pair<T1, T2> const &p, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
69 return std::make_pair(reduce(p.first, c, root, all, op), reduce(p.second, c, root, all, op));
70 }
