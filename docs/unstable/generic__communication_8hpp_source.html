46 // Type trait to check if a type is a std::vector.
47 template <typename T> inline constexpr bool is_std_vector = false;
48
49 // Spezialization of is_std_vector for std::vector<T>.
50 template <typename T> inline constexpr bool is_std_vector<std::vector<T>> = true;
51
52 // Convert an object of type V to an object of type T.
53 template <typename T, typename V> T convert(V v) {
54 if constexpr (is_std_vector<T>) {
55 T res;
56 res.reserve(v.size());
57 for (auto &x : v) res.emplace_back(convert<typename T::value_type>(std::move(x)));
58 return res;
59 } else
60 return T{std::move(v)};
61 }

66 * @brief Generic MPI broadcast.
67 *
68 * @details If mpi::has_env is true, this function calls the specialized `mpi_broadcast` function for the given
69 * object, otherwise it does nothing.
70 *
71 * @tparam T Type to be broadcasted.
72 * @param x Object to be broadcasted.
73 * @param c mpi::communicator.
74 * @param root Rank of the root process.
75 */
76 template <typename T> [[gnu::always_inline]] void broadcast(T &x, communicator c = {}, int root = 0) {
77 static_assert(not std::is_const_v<T>, "mpi::broadcast cannot be called on const objects");
78 if (has_env) mpi_broadcast(x, c, root);

81 /**
82 * @brief Generic MPI reduce.
83 *
84 * @details If mpi::has_env is true or if the return type of the specialized `mpi_reduce` is lazy, this function calls
85 * the specialized `mpi_reduce` function for the given object. Otherwise, it simply converts the input object to the
86 * output type `mpi_reduce` would return.
87 *
88 * @tparam T Type to be reduced.
89 * @param x Object to be reduced.
90 * @param c mpi::communicator.
91 * @param root Rank of the root process.
92 * @param all Should all processes receive the result of the reduction.
93 * @param op `MPI_Op` used in the reduction.
94 * @return The result of the specialized `mpi_reduce` call.
95 */
96 template <typename T>
97 [[gnu::always_inline]] inline decltype(auto) reduce(T &&x, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
98 // return type of mpi_reduce
99 using r_t = decltype(mpi_reduce(std::forward<T>(x), c, root, all, op));
100 if constexpr (is_mpi_lazy<r_t>) {
101 return mpi_reduce(std::forward<T>(x), c, root, all, op);
102 } else {
103 if (has_env)
104 return mpi_reduce(std::forward<T>(x), c, root, all, op);
105 else
106 return detail::convert<r_t>(std::forward<T>(x));
107 }
108 }

206 /**
207 * @brief Implementation of an MPI broadcast for types that have a corresponding MPI datatype, i.e. for which a
208 * specialization of mpi::mpi_type has been defined.
209 *
210 * @details It throws an exception in case a call to the MPI C library fails.
211 *
212 * @tparam T Type to be broadcasted.
213 * @param x Object to be broadcasted.
214 * @param c mpi::communicator.
215 * @param root Rank of the root process.
216 */
217 template <typename T>
218 requires(has_mpi_type<T>)
219 void mpi_broadcast(T &x, communicator c = {}, int root = 0) {
220 check_mpi_call(MPI_Bcast(&x, 1, mpi_type<T>::get(), root, c.get()), "MPI_Bcast");
221 }

223 /**
224 * @brief Implementation of an MPI reduce for types that have a corresponding MPI datatype, i.e. for which a
225 * specialization of mpi::mpi_type has been defined.
226 *
227 * @details It throws an exception in case a call to the MPI C library fails.
228 *
229 * @tparam T Type to be reduced.
230 * @param x Object to be reduced.
231 * @param c mpi::communicator.
232 * @param root Rank of the root process.
233 * @param all Should all processes receive the result of the reduction.
234 * @param op `MPI_Op` used in the reduction.
235 * @return The result of the reduction.
236 */
237 template <typename T>
238 requires(has_mpi_type<T>)
239 T mpi_reduce(T const &x, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
240 T b;
241 auto d = mpi_type<T>::get();
242 if (!all)
243 // old MPI implementations may require a non-const send buffer
244 check_mpi_call(MPI_Reduce(const_cast<T *>(&x), &b, 1, d, op, root, c.get()), "MPI_Reduce"); // NOLINT
245 else
246 check_mpi_call(MPI_Allreduce(const_cast<T *>(&x), &b, 1, d, op, c.get()), "MPI_Allreduce"); // NOLINT
247 return b;
248 }

250 /**
251 * @brief Implementation of an in-place MPI reduce for types that have a corresponding MPI datatype, i.e. for which
252 * a specialization of mpi::mpi_type has been defined.
253 *
254 * @details It throws an exception in case a call to the MPI C library fails.
255 *
256 * @tparam T Type to be reduced.
257 * @param x Object to be reduced.
258 * @param c mpi::communicator.
259 * @param root Rank of the root process.
260 * @param all Should all processes receive the result of the reduction.
261 * @param op `MPI_Op` used in the reduction.
262 */
263 template <typename T>
264 requires(has_mpi_type<T>)
265 void mpi_reduce_in_place(T &x, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
266 if (!all)
267 check_mpi_call(MPI_Reduce((c.rank() == root ? MPI_IN_PLACE : &x), &x, 1, mpi_type<T>::get(), op, root, c.get()), "MPI_Reduce");
268 else
269 check_mpi_call(MPI_Allreduce(MPI_IN_PLACE, &x, 1, mpi_type<T>::get(), op, c.get()), "MPI_Allreduce");
270 }

272 /**
273 * @brief Checks if a given object is equal across all ranks in the given communicator.
274 *
275 * @details It requires that there is a specialized `mpi_reduce` for the given type `T` and that it is equality
276 * comparable as well as default constructible.
277 *
278 * It makes two calls to simplemc::mpi::all_reduce, one with `MPI_MIN` and the other with `MPI_MAX`, and compares
279 * their results.
280 *
281 * @note `MPI_MIN` and `MPI_MAX` need to make sense for the given type `T`.
282 *
283 * @tparam T Type to be checked.
284 * @param x Object to be equality compared.
285 * @param c mpi::communicator.
286 * @return If the given object is equal on all ranks, it returns true. Otherwise, it returns false.
287 */
288 template <typename T> bool all_equal(T &&x, communicator c = {}) {
289 if (!has_env) return true;
290 auto min_obj = all_reduce(std::forward<T>(x), c, MPI_MIN);
291 auto max_obj = all_reduce(std::forward<T>(x), c, MPI_MAX);
292 return min_obj == max_obj;
293 }
