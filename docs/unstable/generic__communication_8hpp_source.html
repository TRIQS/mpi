44 // Type trait to check if a type is a std::vector.
45 template <typename T> inline constexpr bool is_std_vector = false;
46
47 // Spezialization of is_std_vector for std::vector<T>.
48 template <typename T> inline constexpr bool is_std_vector<std::vector<T>> = true;
49
50 // Convert an object of type V to an object of type T.
51 template <typename T, typename V> T convert(V v) {
52 if constexpr (is_std_vector<T>) {
53 T res;
54 res.reserve(v.size());
55 for (auto &x : v) res.emplace_back(convert<typename T::value_type>(std::move(x)));
56 return res;
57 } else
58 return T{std::move(v)};
59 }

63 /**
64 * @brief Generic MPI broadcast.
65 *
66 * @details If mpi::has_env is true, this function calls the specialized `mpi_broadcast` function for the given object,
67 * otherwise it does nothing.
68 *
69 * @tparam T Type to be broadcasted.
70 * @param x Object to be broadcasted.
71 * @param c mpi::communicator.
72 * @param root Rank of the root process.
73 */
74 template <typename T> [[gnu::always_inline]] void broadcast(T &x, communicator c = {}, int root = 0) {
75 static_assert(not std::is_const_v<T>, "mpi::broadcast cannot be called on const objects");
76 if (has_env) mpi_broadcast(x, c, root);
77 }

79 /**
80 * @brief Generic MPI reduce.
81 *
82 * @details If mpi::has_env is true or if the return type of the specialized `mpi_reduce` is lazy, this function calls
83 * the specialized `mpi_reduce` function for the given object. Otherwise, it simply converts the input object to the
84 * output type `mpi_reduce` would return.
85 *
86 * @tparam T Type to be reduced.
87 * @param x Object to be reduced.
88 * @param c mpi::communicator.
89 * @param root Rank of the root process.
90 * @param all Should all processes receive the result of the reduction.
91 * @param op `MPI_Op` used in the reduction.
92 * @return The result of the specialized `mpi_reduce` call.
93 */
94 template <typename T>
95 [[gnu::always_inline]] inline decltype(auto) reduce(T &&x, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
96 // return type of mpi_reduce
97 using r_t = decltype(mpi_reduce(std::forward<T>(x), c, root, all, op));
98 if constexpr (is_mpi_lazy<r_t>) {
99 return mpi_reduce(std::forward<T>(x), c, root, all, op);
100 } else {
101 if (has_env)
102 return mpi_reduce(std::forward<T>(x), c, root, all, op);
103 else
104 return detail::convert<r_t>(std::forward<T>(x));
105 }
106 }

204 /**
205 * @brief Implementation of an MPI broadcast for types that have a corresponding MPI datatype, i.e. for which
206 * a specialization of mpi::mpi_type has been defined.
207 *
208 * @tparam T Type to be broadcasted.
209 * @param x Object to be broadcasted.
210 * @param c mpi::communicator.
211 * @param root Rank of the root process.
212 */
213 template <typename T>
214 requires(has_mpi_type<T>)
215 void mpi_broadcast(T &x, communicator c = {}, int root = 0) {
216 MPI_Bcast(&x, 1, mpi_type<T>::get(), root, c.get());
217 }

219 /**
220 * @brief Implementation of an MPI reduce for types that have a corresponding MPI datatype, i.e. for which
221 * a specialization of mpi::mpi_type has been defined.
222 *
223 * @tparam T Type to be reduced.
224 * @param x Object to be reduced.
225 * @param c mpi::communicator.
226 * @param root Rank of the root process.
227 * @param all Should all processes receive the result of the reduction.
228 * @param op `MPI_Op` used in the reduction.
229 * @return The result of the reduction.
230 */
231 template <typename T>
232 requires(has_mpi_type<T>)
233 T mpi_reduce(T const &x, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
234 T b;
235 auto d = mpi_type<T>::get();
236 if (!all)
237 // old MPI implementations may require a non-const send buffer
238 MPI_Reduce(const_cast<T *>(&x), &b, 1, d, op, root, c.get()); // NOLINT
239 else
240 MPI_Allreduce(const_cast<T *>(&x), &b, 1, d, op, c.get()); // NOLINT
241 return b;
242 }

244 /**
245 * @brief Implementation of an in-place MPI reduce for types that have a corresponding MPI datatype, i.e. for which
246 * a specialization of mpi::mpi_type has been defined.
247 *
248 * @tparam T Type to be reduced.
249 * @param x Object to be reduced.
250 * @param c mpi::communicator.
251 * @param root Rank of the root process.
252 * @param all Should all processes receive the result of the reduction.
253 * @param op `MPI_Op` used in the reduction.
254 */
255 template <typename T>
256 requires(has_mpi_type<T>)
257 void mpi_reduce_in_place(T &x, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
258 if (!all)
259 MPI_Reduce((c.rank() == root ? MPI_IN_PLACE : &x), &x, 1, mpi_type<T>::get(), op, root, c.get());
260 else
261 MPI_Allreduce(MPI_IN_PLACE, &x, 1, mpi_type<T>::get(), op, c.get());
262 }
