33 /**
34 * @ingroup utilities
35 * @brief Get the length of the i<sup>th</sup> subrange after splitting the integer range `[0, end)` evenly across n subranges.
36 *
37 * @param end End of the integer range `[0, end)`.
38 * @param n Number of subranges.
39 * @param i Index of the subrange of interest.
40 * @return Length of the i<sup>th</sup> subrange.
41 */
42 [[nodiscard]] inline long chunk_length(long end, int n, int i) {
43 auto [node_begin, node_end] = itertools::chunk_range(0, end, n, i);
44 return node_end - node_begin;
45 }
46
47 /**
48 * @ingroup utilities
49 * @brief Divide a given range as evenly as possible across the MPI processes in a communicator and get the subrange
50 * assigned to the calling process.
51 *
52 * @tparam R Range type.
53 * @param rg Range to divide.
54 * @param c mpi::communicator.
55 * @return An itertools::sliced range assigned to the calling process.
56 */
57 template <typename R> [[nodiscard]] auto chunk(R &&rg, communicator c = {}) {
58 auto total_size = itertools::distance(std::cbegin(rg), std::cend(rg));
59 auto [start_idx, end_idx] = itertools::chunk_range(0, total_size, c.size(), c.rank());
60 return itertools::slice(std::forward<R>(rg), start_idx, end_idx);
61 }
