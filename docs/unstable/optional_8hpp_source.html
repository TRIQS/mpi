53 template <typename T> void mpi_broadcast(std::optional<T> &opt, communicator c = {}, int root = 0) {
54 bool has_val = opt.has_value();
55 broadcast(has_val, c, root);
56
57 if (has_val) {
58 if (!opt.has_value()) opt.emplace();
59 broadcast(*opt, c, root);
60 } else {
61 opt.reset();
62 }
63 }

82 template <typename T>
83 std::optional<T> mpi_reduce(std::optional<T> const &opt, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
84 // Verify consistency: sum of has_value should be 0 or c.size()
85 int has_val = opt.has_value() ? 1 : 0;
86 int total = mpi::all_reduce(has_val, c, MPI_SUM);
87 if (total != 0 && total != c.size()) {
88 throw std::runtime_error("mpi::reduce for std::optional requires all ranks to have consistent has_value state");
89 }
90
91 if (opt.has_value()) return reduce(*opt, c, root, all, op);
92 else return {};
93 }

114 template <typename T1, typename T2>
115 void mpi_reduce_into(std::optional<T1> const &opt_in, std::optional<T2> &opt_out, communicator c = {}, int root = 0, bool all = false,
116 MPI_Op op = MPI_SUM) {
117 // Verify consistency
118 int has_val = opt_in.has_value() ? 1 : 0;
119 int total = mpi::all_reduce(has_val, c, MPI_SUM);
120 if (total != 0 && total != c.size()) {
121 throw std::runtime_error("mpi::reduce_into for std::optional requires all ranks to have consistent has_value state");
122 }
123
124 if (opt_in.has_value()) {
125 if (!opt_out.has_value()) opt_out.emplace();
126 reduce_into(*opt_in, *opt_out, c, root, all, op);
127 } else {
128 opt_out.reset();
129 }
