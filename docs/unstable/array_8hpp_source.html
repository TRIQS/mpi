43 * @details It simply calls mpi::broadcast_range with the input array.

45 * @tparam T Value type of the array.

53 /**
54 * @brief Implementation of an in-place MPI reduce for a std::array.
55 *
56 * @details It simply calls mpi::reduce_in_place_range with the given input array.
57 *
58 * @tparam T Value type of the array.
59 * @tparam N Size of the array.
60 * @param arr std::array to reduce.
61 * @param c mpi::communicator.
62 * @param root Rank of the root process.
63 * @param all Should all processes receive the result of the reduction.
64 * @param op `MPI_Op` used in the reduction.
65 */
66 template <typename T, std::size_t N>
67 void mpi_reduce_in_place(std::array<T, N> &arr, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
68 reduce_in_place_range(arr, c, root, all, op);
69 }

71 /**
72 * @brief Implementation of an MPI reduce for a std::array.
73 *
74 * @details It simply calls mpi::reduce_range with the given input array and an empty array of the same size.
75 *
76 * @tparam T Value type of the array.
77 * @tparam N Size of the array.
78 * @param arr std::array to reduce.
79 * @param c mpi::communicator.
80 * @param root Rank of the root process.
81 * @param all Should all processes receive the result of the reduction.
82 * @param op `MPI_Op` used in the reduction.
83 * @return std::array containing the result of each individual reduction.
84 */
85 template <typename T, std::size_t N>
86 auto mpi_reduce(std::array<T, N> const &arr, communicator c = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) {
87 std::array<regular_t<T>, N> res{};
88 reduce_range(arr, res, c, root, all, op);
89 return res;
90 }
