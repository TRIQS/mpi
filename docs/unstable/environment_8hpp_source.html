35 /**
36 * @brief Check if MPI has been initialized.
37 * @return True if `MPI_Init` has been called, false otherwise.
38 */
39 [[nodiscard]] inline bool is_initialized() noexcept {
40 int flag = 0;
41 MPI_Initialized(&flag);
42 return flag;
43 }
44
45 /**
46 * @brief Boolean variable that is true, if one of the environment variables `OMPI_COMM_WORLD_RANK`,
47 * `PMI_RANK`, `CRAY_MPICH_VERSION` or `FORCE_MPI_INIT` is set, false otherwise.
48 *
49 * @details The environment variables are set, when a program is executed with `mpirun` or `mpiexec`.
50 */
51 static const bool has_env = []() {
52 if (std::getenv("OMPI_COMM_WORLD_RANK") != nullptr or std::getenv("PMI_RANK") != nullptr or std::getenv("CRAY_MPICH_VERSION") != nullptr
53 or std::getenv("FORCE_MPI_INIT") != nullptr)
54 return true;
55 else
56 return false;
57 }();
58
59 /**
60 * @brief RAII class to initialize and finalize MPI.
61 *
62 * @details Calls `MPI_Init` upon construction and `MPI_Finalize` upon destruction i.e. when the environment object goes out of scope.
63 * If mpi::has_env is false, this struct does nothing.
64 */
65 struct environment {
66 /**
67 * @brief Construct a new mpi environment object by calling `MPI_Init`.
68 *
69 * @details Checks first if the program is run with an MPI runtime environment and if it has not been
70 * initialized before to avoid errors.
71 *
72 * @param argc Number of command line arguments.
73 * @param argv Command line arguments.
74 */
75 environment(int argc, char *argv[]) { // NOLINT (C-style array is wanted here)
76 if (has_env && !is_initialized()) MPI_Init(&argc, &argv);
77 }
78
79 /**
80 * @brief Destroy the mpi environment object by calling `MPI_Finalize`.
81 *
82 * @details Checks first if the program is run with an MPI runtime environment. Called automatically when the environment
83 * object goes out of scope.
84 */
85 ~environment() {
86 if (has_env) MPI_Finalize();
87 }
88 };
