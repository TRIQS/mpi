190 void barrier(long poll_msec = 1) const {
191 if (has_env) {
192 if (poll_msec == 0) {
193 check_mpi_call(MPI_Barrier(comm_), "MPI_Barrier");
194 } else {
195 MPI_Request req{};
196 int flag = 0;
197 check_mpi_call(MPI_Ibarrier(comm_, &req), "MPI_Ibarrier");
198 while (!flag) {
199 check_mpi_call(MPI_Test(&req, &flag, MPI_STATUS_IGNORE), "MPI_Test");
200 usleep(poll_msec * 1000);
201 }
202 }
203 }
